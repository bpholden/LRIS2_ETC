```sh
pip install .        # or: pip install -e .  for development
```

```sh
lris2-etc --help
lris2-etc -m 22 -t 1200 -T starb1 --red_grism R400 --blue_grism B600
lris2-compare        # LRIS-2 compared with the current LRIS
```
